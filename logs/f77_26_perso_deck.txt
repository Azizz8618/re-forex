      COMMON /FILMAP/ FM
      INTEGER NM(2), LUN, NZ, IST, OP, EX, N, I

    1 FM(I) = 0
      NM(1) = 0
      NM(2) = 0
      LUN = 41
      NZ = 8
      N = 1024
      DO 5 I=1,8
    5 BUF(I) = I
C ---- CREATE + OPEN ----
      CALL ZFOPEN(NM,LUN,NZ,IST)
      PRINT 100, 1, IST
      PRINT 100, 2, FM(4)
      PRINT 100, 3, FM(8)
C ---- WRITE 2 ZONES ----
      CALL ZFWRIT(LUN,BUF,N,IST)
      PRINT 100, 4, IST
      PRINT 100, 5, FM(8)
      CALL ZFWRIT(LUN,BUF,N,IST)
      PRINT 100, 6, IST
      PRINT 100, 7, FM(8)
C ---- READ THEM BACK ----
      FM(8) = 0
      CALL ZFREAD(LUN,BUF,N,IST)
      PRINT 100, 8, IST
      PRINT 100, 9, FM(248)
      CALL ZFREAD(LUN,BUF,N,IST)
      PRINT 100, 10, IST
      PRINT 100, 11, FM(248)
